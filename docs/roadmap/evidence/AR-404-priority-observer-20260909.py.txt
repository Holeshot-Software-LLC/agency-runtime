FOLLOWUP = CASES[1]['request']
def matches_packet(value):
    return 'critic_contract' in value and (value.get('request') == ASK or (value.get('request') == FOLLOWUP and 'average' in json.dumps(value).lower()))
PROFILE = 'agency-recruiter-critic'
MODEL = 'task-agency-critic-v2'
LIMIT = 2 * 1024 * 1024
HOP = {'connection', 'keep-alive', 'proxy-authenticate', 'proxy-authorization', 'te', 'trailer', 'transfer-encoding', 'upgrade', 'content-length', 'host'}


def public_response(data):
    try:
        value=json.loads(data)
        return {'model':value.get('model'), 'choices':[{'message':{'content':row.get('message',{}).get('content')},'finish_reason':row.get('finish_reason')} for row in value.get('choices',[])]}
    except (ValueError,TypeError,AttributeError):
        return {'capture_error':'non_json_public_response'}

class Observer(BaseHTTPRequestHandler):
    def log_message(self, *args):
        pass
    def do_POST(self):
        size = int(self.headers.get('Content-Length', '0'))
        if not 0 < size <= LIMIT:
            self.send_error(413)
            return
        body = self.rfile.read(size)
        packet = None
        try:
            request = json.loads(body)
            if request.get('model') == MODEL:
                for message in request.get('messages', []):
                    if message.get('role') == 'user':
                        value = json.loads(message.get('content', '{}'))
                        if matches_packet(value):
                            packet = value
        except (ValueError, TypeError, AttributeError):
            pass
        conn = http.client.HTTPConnection(*self.server.upstream, timeout=65)
        started = time.time()
        try:
            headers = {k:v for k,v in self.headers.items() if k.lower() not in HOP}
            conn.request('POST', self.path, body=body, headers=headers)
            response = conn.getresponse()
            data = response.read(LIMIT + 1)
            if len(data) > LIMIT:
                raise ValueError('bounded_response_exceeded')
            if packet is not None and self.server.capture_dir is not None:
                record = {'started_at':started, 'ended_at':time.time(), 'request_sha256':hashlib.sha256(body).hexdigest(), 'response_sha256':hashlib.sha256(data).hexdigest(), 'http_status':response.status, 'critic_packet':packet, 'response_text':json.dumps(public_response(data),ensure_ascii=False)}
                try:
                    target = self.server.capture_dir / (str(time.monotonic_ns())+'.json')
                    with target.open('x') as f:
                        json.dump(record, f, indent=2)
                except OSError:
                    self.server.errors.append('capture_write_failed')
            self.send_response(response.status)
            for k,v in response.getheaders():
                if k.lower() not in HOP:
                    self.send_header(k,v)
            self.send_header('Content-Length', str(len(data)))
            self.end_headers()
            self.wfile.write(data)
        except Exception as exc:
            self.server.errors.append(type(exc).__name__)
            self.send_error(502)
        finally:
            conn.close()

def server(upstream, capture_dir):
    instance = ThreadingHTTPServer(('127.0.0.1', 0), Observer)
    instance.upstream = upstream
    instance.capture_dir = capture_dir
    instance.errors = []
    threading.Thread(target=instance.serve_forever, daemon=True).start()
    return instance

def atomic_raw(path, raw):
    descriptor, name = tempfile.mkstemp(prefix='.ar414-',dir=path.parent)
    try:
        with os.fdopen(descriptor,'wb') as f:
            f.write(raw);f.flush();os.fsync(f.fileno())
        os.replace(name,path)
    finally:
        if os.path.exists(name):os.unlink(name)
